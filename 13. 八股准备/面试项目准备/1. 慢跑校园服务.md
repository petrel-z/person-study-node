## Q2. 
(基于文件上传) 你说做了断点续传，如果用户传了一半关闭小程序，下次进来怎么知道传到哪了？
### 大文件上传解决了哪些问题：
1. 用户需要上传的文件太大，上传成功率低，速度慢，易超时
2. 可能上传到一半，上传失败了，需要重新上传
3. 上传请求占用宽带过高
4. 服务器处理时压力大

如何解决这些问题？
分片上传、秒传、断点续传、并发控制、重试机制

如何控制上传的顺序？
只需要index索引控制即可，后端会根据文件的hash值，创建一个对应的切片，按照前端传递的索引值进行存储。

如何知道哪些文件上传成功了？
通过init接口，后端返回已经上传的切片有哪些，前端将已上传的切片进行过滤。

如何上传一部分后，由于网络波动导致上传暂停，重新上传不必再上传已经上传过的文件？（断点续传）
跟上面一致

如何做到并发控制？
Promise 队列 + 运行数控制

重传机制如何实现？
用一个变量控制上传次数，当请求上传接口失败后，执行重新上传

单个分片上传可封装：上传请求、失败重传、状态更新、进度统计

### 双token机制的，并发控制
```js
// utils/request.js  封装axios请求实例
import axios from 'axios';
import { 
  refreshTokenPromise, 
  setToken, 
  getToken, 
  removeToken, 
  resetRefreshPromise 
} from './token';
import { navigateToLogin } from './router'; // 自定义跳登录方法（如vue-router/react-router）

// 创建axios实例
const service = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL, // 接口基础地址
  timeout: 5000,
});

// ************************ 1. 请求拦截器：统一添加accessToken ************************
service.interceptors.request.use(
  (config) => {
    const { accessToken } = getToken();
    // 业务请求统一携带accessToken（刷新Token接口除外）
    if (accessToken && !config.url.includes('/refresh-token')) {
      config.headers.Authorization = `Bearer ${accessToken}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// ************************ 2. 封装刷新Token的方法：返回Promise ************************
// 仅用于刷新accessToken，请求头携带refreshToken
const refreshToken = async () => {
  const { refreshToken } = getToken();
  try {
    const res = await axios({
      url: `${import.meta.env.VITE_API_BASE_URL}/refresh-token`,
      method: 'POST',
      headers: {
        Authorization: `Bearer ${refreshToken}`, // 用refreshToken换accessToken
      },
    });
    const { newAccessToken, newRefreshToken } = res.data;
    // 刷新成功：更新本地双Token
    setToken(newAccessToken, newRefreshToken);
    return newAccessToken; // 返回新的accessToken，供后续请求重试
  } catch (err) {
    // 刷新失败：说明refreshToken也过期，清除Token并抛错
    removeToken();
    navigateToLogin(); // 跳转到登录页
    throw new Error('refreshToken过期，请重新登录');
  }
};

// ************************ 3. 响应拦截器：统一处理401，控制并发刷新 ************************
service.interceptors.response.use(
  (response) => response.data, // 正常响应：直接返回数据
  async (error) => {
    const originalRequest = error.config; // 保存原始请求配置（用于后续重试）
    // 判定条件：1. 响应状态为401  2. 不是刷新Token接口本身  3. 未被重复重试过（避免死循环）
    if (
      error.response?.status === 401 &&
      !originalRequest.url.includes('/refresh-token') &&
      !originalRequest._retry
    ) {
      originalRequest._retry = true; // 标记为已重试，避免死循环

      try {
        // 核心：通过单例Promise控制，仅发起一次刷新Token请求
        if (!refreshTokenPromise) {
          // 无正在进行的刷新请求：发起请求并赋值给全局单例
          refreshTokenPromise = refreshToken();
        }
        // 所有并发的401请求，都等待同一个刷新Token的Promise结果
        const newAccessToken = await refreshTokenPromise;

        // 刷新成功：用新的accessToken更新原始请求的请求头，并重试原始请求
        originalRequest.headers.Authorization = `Bearer ${newAccessToken}`;
        return service(originalRequest); // 重试请求，返回新的Promise
      } catch (err) {
        return Promise.reject(err); // 刷新失败：直接抛错，上层业务处理
      } finally {
        // 关键：刷新完成（成功/失败）后，重置全局单例Promise为null
        // 确保后续的Token过期请求，能正常触发新的刷新
        resetRefreshPromise();
      }
    }

    // 非401错误，或刷新失败：直接抛错
    return Promise.reject(error);
  }
);

export default service;
```

### 并发控制、失败重传

```js
/**
 * 分片并发上传
 * @param {Array} chunks - 切片列表（含chunk、index、hash等）
 * @param {String} fileMd5 - 文件整体Hash
 * @param {Number} maxConcurrent - 最大并发数，默认3
 * @returns {Promise} 所有切片上传完成的Promise
 */
const uploadChunksWithConcurrent = async (chunks, fileMd5, maxConcurrent = 3) => {
  const queue = []; // 待上传队列
  let running = 0; // 正在运行的上传数
  // 初始化分片状态数组
  const chunkStatus = new Array(chunks.length).fill(false);

  // 单个切片上传方法（含失败重传，最大3次）
  const uploadSingleChunk = async (chunk, retry = 3) => {
    const { chunk: blob, index, hash: chunkHash } = chunk;
    try {
      const formData = new FormData();
      formData.append('chunk', blob);
      formData.append('fileMd5', fileMd5);
      formData.append('chunkIndex', index);
      formData.append('chunkHash', chunkHash);
      formData.append('totalChunks', chunks.length);

      // 上传请求
      await axios.post('/api/upload-chunk', formData, {
        onUploadProgress: (e) => {
          // 实时更新单个切片进度（可选）
          chunk.progress = (e.loaded / e.total) * 100;
        },
      });
      // 上传成功，更新状态
      chunkStatus[index] = true;
      console.log(`切片${index}上传成功`);
    } catch (err) {
      if (retry > 0) {
        console.log(`切片${index}上传失败，剩余重传次数${retry-1}，开始重传`);
        return uploadSingleChunk(chunk, retry - 1); // 递归重传
      }
      throw new Error(`切片${index}上传失败，已达最大重传次数`);
    }
  };

  // 队列执行方法：取出下一个切片上传
  const runNext = async () => {
    if (queue.length === 0 || running >= maxConcurrent) return;
    running++;
    const nextChunk = queue.shift();
    try {
      await uploadSingleChunk(nextChunk);
    } finally {
      running--;
      runNext(); // 完成后，立即执行下一个
    }
  };

  // 遍历所有切片，加入队列或直接上传
  for (const chunk of chunks) {
    if (running < maxConcurrent) {
      running++;
      try {
        await uploadSingleChunk(chunk);
      } finally {
        running--;
        runNext();
      }
    } else {
      queue.push(chunk);
    }
  }

  // 等待队列中所有切片上传完成
  while (running > 0 || queue.length > 0) {
    await new Promise(resolve => setTimeout(resolve, 100));
  }

  // 校验所有切片是否上传成功
  const allSuccess = chunkStatus.every(Boolean);
  if (!allSuccess) throw new Error('部分切片上传失败，请重新尝试');
  console.log('所有切片上传成功，准备请求合并');
  return chunkStatus;
};
```
