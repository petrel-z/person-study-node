// ✅ 手写通用浅拷贝函数（兼容数组+对象）
function shallowClone(target) {
  // 1. 如果不是引用类型，直接返回原值（基本类型无需拷贝）
  if (typeof target !== 'object' || target === null) {
    return target;
  }
  // 2. 判断是数组还是对象，创建对应的空容器
  const newObj = Array.isArray(target) ? [] : {};
  // 3. 遍历目标的【第一层属性】，逐个赋值
  for (let key in target) {
    // 只拷贝自身属性，不拷贝原型链上的属性
    if (target.hasOwnProperty(key)) {
      newObj[key] = target[key];
    }
  }
  // 4. 返回新的浅拷贝数据
  return newObj;
}

// 测试使用
const obj = {name: '张三', hobby: ['篮球']};
const obj1 = shallowClone(obj);

// ✅ 手写通用递归深拷贝函数（兼容数组+对象，面试标准答案，必背！）
function deepClone(target) {
  // 1. 处理基本类型 + null：直接返回原值
  if (typeof target !== 'object' || target === null) {
    return target;
  }
  // 2. 处理数组/对象：创建新的空容器
  const newObj = Array.isArray(target) ? [] : {};
  // 3. 遍历所有属性，递归拷贝
  for (let key in target) {
    // 只拷贝自身属性，排除原型链属性
    if (target.hasOwnProperty(key)) {
      // 核心：如果属性值是引用类型，递归调用deepClone；否则直接赋值
      newObj[key] = deepClone(target[key]);
    }
  }
  // 4. 返回新的深拷贝数据
  return newObj;
}

// 测试使用（多层嵌套，完美深拷贝 ✔️）
const obj = {name: '张三', hobby: ['篮球', ['听歌']], info: {age:18}};
const obj1 = deepClone(obj);
obj1.hobby[1].push('游戏');
obj1.info.age = 20;
console.log(obj); // 源数据完全不变 ✔️

// 优化版

function deepClone(target, map = new WeakMap()) {
  if (typeof target !== 'object' || target === null) return target;
  if (map.has(target)) return map.get(target); // 处理循环引用
  const newObj = Array.isArray(target) ? [] : {};
  map.set(target, newObj); // 存储已拷贝的对象
  for (let key in target) {
    if (target.hasOwnProperty(key)) {
      newObj[key] = deepClone(target[key], map);
    }
  }
  return newObj;
}
