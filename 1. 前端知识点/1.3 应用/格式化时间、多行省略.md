## 工具函数封装
```ts
// utils/time.ts
export const formatRelativeTime = (date: string) => {
  const timeRanges = [
    { threshold: 0, format: () => '今天' },
    { threshold: 1, format: () => '昨天' },
    { threshold: 2, format: () => '前天' },
    { threshold: 7, format: (days) => `${days}天前` },
    { threshold: 30, format: (days) => `${Math.floor(days/7)}周前` },
    { threshold: 365, format: (days) => `${Math.floor(days/30)}个月前` },
    { threshold: Infinity, format: (days) => `${Math.floor(days/365)}年前` }
  ]
  
  const diffDays = Math.floor((Date.now() - new Date(date).getTime()) / (1000*60*60*24))
  const range = timeRanges.find(r => diffDays < r.threshold) || timeRanges[timeRanges.length-1]
  return range.format(diffDays)
}
```


## if-else
```ts
const getDate = computed(() => {

const time = dayjs(props.notice.date).valueOf();

const now = dayjs().valueOf();

const diff = now - time;

const diffDay = Math.floor(diff / (1000 * 60 * 60 * 24));

  

// 使用范围判断而非精确匹配

if (diffDay === 0) {

return '今天';

} else if (diffDay === 1) {

return '昨天';

} else if (diffDay === 2) {

return '前天';

} else if (diffDay >= 3 && diffDay < 7) {

return diffDay + '天前';

} else if (diffDay >= 7 && diffDay < 30) {

const weeks = Math.floor(diffDay / 7);

return weeks === 1 ? '1周前' : weeks + '周前';

} else if (diffDay >= 30 && diffDay < 365) {

const months = Math.floor(diffDay / 30);

return months === 1 ? '1个月前' : months + '个月前';

} else if (diffDay >= 365) {

const years = Math.floor(diffDay / 365);

return years === 1 ? '1年前' : years + '年前';

} else {

return '刚刚';

}

});
```
