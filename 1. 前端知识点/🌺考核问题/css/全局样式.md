### css预处理器实现方式：
- 安装css预处理器
- 创建全局样式文件`src/styles/variables.scss`
```scss
// scss文件中
// 全局变量
$primary-color: #42b983;
$font-size-base: 14px;
$border-radius: 4px;

```
- 在vite.config.ts中配置全局变量
```javascript
// vite.config.js
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'

export default defineConfig({
  plugins: [vue()],
  css: {
    preprocessorOptions: {
      scss: {
        // 引入全局变量文件（多个文件用逗号分隔）
        additionalData: `@import "./src/styles/variables.scss";`
      }
    }
  }
})


```

```vue
// 任意组件使用
<style scoped lang="scss">
.button {
  background-color: $primary-color;  // 直接使用全局变量
  font-size: $font-size-base;
  border-radius: $border-radius;
}
</style>
```


### 组件库的全局样式覆盖
以ant design为例
1. 安装less
2. 创建自定义主题文件src/styles/antd-theme.less
```less
// 引入 Ant Design 基础样式（必须）
@import 'ant-design-vue/dist/antd.less';

// 覆盖内置变量（完整变量列表见下方链接）
@primary-color: #1890ff; // 全局主色（默认蓝色）
@success-color: #52c41a; // 成功色
@warning-color: #faad14; // 警告色
@error-color: #f5222d; // 错误色
@font-size-base: 14px; // 基础字体大小
@border-radius-base: 4px; // 基础边框半径
@border-color-base: #d9d9d9; // 基础边框颜色
@text-color: #333333; // 主要文本颜色
```
3. 配置项目 （vite.config.ts)
```javascript
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'

export default defineConfig({
  plugins: [vue()],
  css: {
    preprocessorOptions: {
      less: {
        // 允许 Less 编译时执行 JavaScript（必须）
        javascriptEnabled: true,
        // 引入自定义主题文件
        additionalData: `@import "./src/styles/antd-theme.less";`
      }
    }
  }
})
```
