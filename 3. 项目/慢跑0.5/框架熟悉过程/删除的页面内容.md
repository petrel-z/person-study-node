<template>

<wd-cell custom-title-class="lh-40px" :title="`列表长度-${index + 1}`">

<template #icon>

<wd-img class="mr-10px" round width="40" height="40" :src="item" />

</template>

</wd-cell>

</view>

</z-paging>

</template>

  

<script setup lang="ts">

import { random } from '@/utils';

// import zPaging from 'z-paging'; // 或实际路径

const pagingRef = ref(null);

'https://picsum.photos/100/100?random=1',

'https://picsum.photos/100/100?random=2',

'https://picsum.photos/100/100?random=3',

'https://picsum.photos/100/100?random=4',

'https://picsum.photos/100/100?random=5',

'https://picsum.photos/100/100?random=6',

'https://picsum.photos/100/100?random=7',

'https://picsum.photos/100/100?random=8',

'https://picsum.photos/100/100?random=9',

'https://picsum.photos/100/100?random=10',

];

// queryList 是分页查询的回调函数，负责加载数据。

function queryList(pageNo: number, pageSize: number) {

// 参数 pageNo 和 pageSize 表示当前页码和每页数据量。

console.log('[ pageNo ] >', pageNo);

console.log('[ pageSize ] >', pageSize);

// 这里的pageNo和pageSize会自动计算好，直接传给服务器即可

// 这里的请求只是演示，请替换成自己的项目的网络请求，并在网络请求回调中通过pagingRef.value.complete(请求回来的数组)将请求结果传给z-paging

/**

模拟网络请求：

使用 setTimeout 模拟 1 秒的延迟。

随机从 urls 数组中选取图片 URL，生成 30 条数据。

*/

// 1秒之后停止刷新动画

const list = [];

for (let i = 0; i < 30; i++)

list.push(urls[random(0, urls.length - 1)]);

  

pagingRef.value?.complete(list);// 将数据传递给 z-paging，调用complete方法来停止刷新动画并更新列表。

}, 1000);
