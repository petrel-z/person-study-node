const formatFileSize = (bytes: number): string => {
	if (typeof bytes !== 'number' || isNaN(bytes) || bytes <= 0) return '-'
	if (bytes < 1024) return `${bytes} B`
	if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
	if (bytes < 1024 * 1024 * 1024) return `${(bytes / 1024 / 1024).toFixed(1)} MB`
	return `${(bytes / 1024 / 1024 / 1024).toFixed(1)} GB`
}

const formatDate = (dateStr: string): string => {
	const date = new Date(dateStr)
	const now = new Date()
	const diffTime = Math.abs(now.getTime() - date.getTime())
	const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24))

	if (diffDays === 0) return '今天'
	if (diffDays === 1) return '昨天'
	if (diffDays < 7) return `${diffDays}天前`
	if (diffDays < 30) return `${Math.floor(diffDays / 7)}周前`
	return dateStr
}

// 优先使用已保存的持久文件，避免重复下载
  const cacheKey = `material_saved_${props.material.id || props.material.url}`;
  const cached: { path: string; updatedAt?: string } | undefined = uni.getStorageSync(cacheKey);
  if (cached?.path) {
    uni.openDocument({
      filePath: cached.path,
      showMenu: true,
      success: () => { },
      fail: () => {
        // 缓存失效则移除并走下载流程
        uni.removeStorage({ key: cacheKey });
        downloadAndPersist();
      }
    });
    return;
  }

  // 若无缓存，下载后持久化并打开
  const downloadAndPersist = () => {
    uni.downloadFile({
      url: props.material.url as string,
      success: (res: { statusCode: number; tempFilePath?: string }) => {
        if (res.statusCode === 200 && res.tempFilePath) {
          uni.saveFile({
            tempFilePath: res.tempFilePath,
            success: (s: { savedFilePath: string }) => {
              uni.setStorage({ key: cacheKey, data: { path: s.savedFilePath, updatedAt: props.material.updatedAt } });
              uni.openDocument({ filePath: s.savedFilePath, showMenu: true });
            },
            fail: () => uni.showToast({ title: '保存失败', icon: 'none' })
          });
        } else {
          uni.showToast({ title: '预览-下载失败', icon: 'none' });
        }
      },
      fail: () => uni.showToast({ title: '预览-下载失败', icon: 'none' })
    });
  };
  downloadAndPersist();
