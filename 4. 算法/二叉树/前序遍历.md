## 递归法：
每一个节点，先是中子树，再是左/右子树
先递归左子树，再递归右子树

```js
/**
 * Definition for a binary tree node.
 * function TreeNode(val, left, right) {
 *     this.val = (val===undefined ? 0 : val)
 *     this.left = (left===undefined ? null : left)
 *     this.right = (right===undefined ? null : right)
 * }
 */
/**
 * @param {TreeNode} root
 * @return {number[]}
 */
var preorderTraversal = function(root) {
    let res = [];
    recursion(root, res);
    return res;
};

let recursion = (root, res) => {
    if(root === null){
        return;
    }
    res.push(root.val);
    recursion(root.left, res);
    recursion(root.right, res);
}
```


迭代法：
数据结构：栈

将根节点压入栈中，弹出
再压入右子树和左子树，弹出
同理

循环条件：栈不为空时循环

```js
/**
 * Definition for a binary tree node.
 * function TreeNode(val, left, right) {
 *     this.val = (val===undefined ? 0 : val)
 *     this.left = (left===undefined ? null : left)
 *     this.right = (right===undefined ? null : right)
 * }
 */
/**
 * @param {TreeNode} root
 * @return {number[]}
 */
var preorderTraversal = function(root) {
    if(root === null){
        return [];
    }
    let res = [];
    let stack = [];
    stack.push(root);
    while(stack.length > 0 ){
        let temp = stack.pop();
        res.push(temp.val);
        if(temp.right){
            stack.push(temp.right)
        }
        if(temp.left){
            stack.push(temp.left);
        } 
    }
    return res;
};
```

## 易错点：
1. 循环条件，与root无关，只有当栈不为空的时候，循环处理栈中的节点
2. 加入栈的时候，先加入右子树的节点， 再加入左子树的节点
避免写成if 和 else if ，因为这会导致只处理一边的子树，应该两个节点都处理，并且右一个先后顺序
