https://leetcode.cn/problems/number-of-islands/description/

深度优先搜索、广度优先搜索

要求：找到被水包围的岛屿的数量

分析问题：
1. 连通的岛屿属于一个岛屿
2. 挨着墙的岛屿中，墙面相当于水

如何找到连通的岛屿？

如何将一块岛屿与另一块岛屿分开算？


1. 采用广度优先搜索
2. 需要判断要访问的数据是否越界
3. dfs是为了将连通的岛屿都访问一遍
4. visited数组是为了记录访问过的岛屿，这样就可以将一块岛屿和另一块岛屿分开了


```js
/**
 * @param {character[][]} grid
 * @return {number}
 */
let res = 0;
var numIslands = function(grid) {
    let m = grid.length;
    let n = grid[0].length;
    let visited = Array.from({length: m}, () => Array.from({length: n }, () => false));
    res = 0;
    for(let i = 0; i < m; i++){
        for(let j = 0; j < n; j++){
            if(grid[i][j] === '1' && !visited[i][j]){
                res++;
                visited[i][j] = true;
                dfs(grid, visited, i, j);
            }
        }
    }
    return res;
};
let dir = [[0,1], [1,0], [0,-1], [-1,0]];
function dfs(grid, visited, i, j){
    for(let k = 0; k < 4; k++){
        nextX = i + dir[k][0];
        nextY = j + dir[k][1];
        if(nextX < 0 || nextY < 0 || nextX >= grid.length || nextY >= grid[0].length){
            continue;
        }
        if(grid[nextX][nextY] === '1' && !visited[nextX][nextY]){
            visited[nextX][nextY] = true;
            dfs(grid, visited, nextX, nextY);
        }
    }
}
```
