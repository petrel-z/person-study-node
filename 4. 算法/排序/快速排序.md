- 时间复杂度：平均 O(nlogN)、最好 O(nlogN)、最坏 O(n²)
- 空间复杂度：O(nlogN)
- In-place 内排序
- 不稳定

# 思路
找一个基准值，遍历其他元素，大的放基准值右边，小的放左边，产生两个数组
将两个子数组继续按照上面的操作，直到数组不能再被分割（结束递归条件）

```js
/**
 * @param {number[]} nums
 * @return {number[]}
 */
var sortArray = function (nums) {
    return quickSort(nums);
};

function quickSort(nums){
    if(nums.length <= 1){
        return nums;
    }
    let index = Math.floor(Math.random() * nums.length);
    let mid = nums[index];
    let arr1 = [], arr2 = [];
    for(let i = 0; i < nums.length; i++){
        if(i === index) {
            continue;
        }
        if(nums[i] <= mid){
            arr1.push(nums[i]);
        }else{
            arr2.push(nums[i]);
        }
    }
    return [...quickSort(arr1), mid , ...quickSort(arr2)];
}
```

# 优化：原地快排

## 思路
1. 随机找基准值，放到数组最右边
2. 根据基准值大小，将数组二分，小的在左，大的在右
	1. 难点：原地调整数组元素位置，让小的在左，大的在右
		1. 思路：相当于用两个指针，指着比基准值大的元素，就移动快指针；指着小的，则将快慢指针元素交换

疑问：怎么数组分成两部分
根据随机索引数，排好序后，返回中间值的所以


```js
/**
 * @param {number[]} nums
 * @return {number[]}
 */
var sortArray = function(nums) {
    quickSort(nums, 0, nums.length - 1);
    return nums;
};

function quickSort(arr, left, right) {

//	if(nums.length <= 1){
//        return;
//    } 错误
    if (left >= right) return; // 区间长度 <= 1，不用排

    // 随机选择一个 pivot，避免退化
    let pivotIndex = Math.floor(Math.random() * (right - left + 1)) + left;
    [arr[pivotIndex], arr[right]] = [arr[right], arr[pivotIndex]]; // swap 到末尾

    // 分区
    let pivotPos = partition(arr, left, right);// 中间元素

    // 递归排序左右两边
    quickSort(arr, left, pivotPos - 1);
    quickSort(arr, pivotPos + 1, right);
}

function partition(arr, left, right) {
    let pivot = arr[right]; // 选最后一个作为基准
    let i = left;

    for (let j = left; j < right; j++) {
        if (arr[j] <= pivot) {
            [arr[i], arr[j]] = [arr[j], arr[i]]; // 交换
            i++;
        }
    }

    // 最后把 pivot 放到正确位置
    [arr[i], arr[right]] = [arr[right], arr[i]];
    return i; // 返回 pivot 的下标
}
```

```js
function findKthLargest(nums, k) {
  const target = nums.length - k; // 找到pivot是第target个元素

  function quickSelect(left, right) {
    if (left === right) return nums[left];
    const pivot = nums[Math.floor(Math.random() * (right - left + 1)) + left];
    let i = left, j = right;
    while (i <= j) {
      while (nums[i] < pivot) i++; // 从左往右，找到大于pivot的值后，停
      while (nums[j] > pivot) j--; // 从右往左，找到小于pivot的值后，停
      if (i <= j) {
        [nums[i], nums[j]] = [nums[j], nums[i]];
        i++; j--;
      }
    }
    if (target <= j) return quickSelect(left, j); // j在pivot左侧
    if (target >= i) return quickSelect(i, right);// i在pivot右侧
    return nums[target];
  }

  return quickSelect(0, nums.length - 1);
}
```  
