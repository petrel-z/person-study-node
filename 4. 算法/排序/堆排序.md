- 时间复杂度：平均 O(nlogn)、最好 O(nlogn)、最坏 O(nlogn)
- 空间复杂度：O(1)
- In-place 内排序
- 不稳定

# 思路
堆排序是利用堆这种数据结构的排序算法
1. 将数组构建成大顶堆（根节点是最大值）
2. 将堆顶元素（最大值）与末尾元素交换
3. 将剩余元素重新调整为大顶堆
4. 重复步骤 2-3，直到堆大小为 1

```js
/**
 * @param {number[]} nums
 * @return {number[]}
 */
var sortArray = function (nums) {
    let n = nums.length;

    // 建堆：从最后一个非叶子节点开始，自底向上调整
    for (let i = Math.floor(n / 2) - 1; i >= 0; i--) {
        heapify(nums, n, i);
    }

    // 排序：将堆顶元素与末尾交换，然后重新调整
    for (let i = n - 1; i > 0; i--) {
        [nums[0], nums[i]] = [nums[i], nums[0]];
        heapify(nums, i, 0);
    }

    return nums;
};

// 调整堆，使以 i 为根的子树成为大顶堆
function heapify(arr, n, i) {
    let largest = i;
    let left = 2 * i + 1;
    let right = 2 * i + 2;

    // 找到 i、左子、右子中的最大值
    if (left < n && arr[left] > arr[largest]) {
        largest = left;
    }
    if (right < n && arr[right] > arr[largest]) {
        largest = right;
    }

    // 如果最大值不是 i，交换并继续调整
    if (largest !== i) {
        [arr[i], arr[largest]] = [arr[largest], arr[i]];
        heapify(arr, n, largest);
    }
}
```

# 完全二叉树的性质
- 节点 i 的左子节点：2*i + 1
- 节点 i 的右子节点：2*i + 2
- 节点 i 的父节点：Math.floor((i - 1) / 2)
- 最后一个非叶子节点：Math.floor(n / 2) - 1

# 特点
1. **不稳定排序**：堆调整过程中可能改变相同元素的相对位置
2. **原地排序**：只需要常数级别的额外空间
3. **适用场景**：TOP-K 问题、优先队列

# 堆排序应用：TOP-K 问题
```js
// 求第 K 大的元素（小顶堆）
function findKthLargest(nums, k) {
    // 构建大小为 k 的小顶堆
    let heap = nums.slice(0, k);

    // 建堆
    for (let i = Math.floor(k / 2) - 1; i >= 0; i--) {
        heapifyMin(heap, k, i);
    }

    // 遍历剩余元素
    for (let i = k; i < nums.length; i++) {
        if (nums[i] > heap[0]) {
            heap[0] = nums[i];
            heapifyMin(heap, k, 0);
        }
    }

    return heap[0];
}

function heapifyMin(arr, n, i) {
    let smallest = i;
    let left = 2 * i + 1;
    let right = 2 * i + 2;

    if (left < n && arr[left] < arr[smallest]) {
        smallest = left;
    }
    if (right < n && arr[right] < arr[smallest]) {
        smallest = right;
    }

    if (smallest !== i) {
        [arr[i], arr[smallest]] = [arr[smallest], arr[i]];
        heapifyMin(arr, n, smallest);
    }
}
```
