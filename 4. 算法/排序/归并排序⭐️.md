# 数组的归并排序

```js
// 二分数组
function split(arr){
    if(arr.length <= 1){
        return [arr, []];
    }
    let len = arr.length;
    let arr1 = arr.slice(0, len / 2);
    let arr2 = arr.slice(len / 2);
    return [arr1, arr2];
}

// 排序并合并
function sortMergeArr(arr1, arr2){
    let i = 0;
    let j = 0;
    let res = [];
    while(i < arr1.length && j < arr2.length){
    // if(arr1[i][0] > arr2[j][0]) 这是二维数组排序（排第一个元素）
        if(arr1[i] > arr2[j]){
            res.push(arr2[j++]);
        }else{
            res.push(arr1[i++]);
        }
    }
    while(i < arr1.length){
        res.push(arr1[i++]);
    }
    while(j < arr2.length){
        res.push(arr2[j++]);
    }
    return res;
}

// 递归
function recursion(arr){
    if(arr.length <= 1){
        return arr;
    }
    let [arr1, arr2] = split(arr);
    let l = recursion(arr1);
    let r = recursion(arr2);
    return sortMergeArr(l, r);
}
```


## 注意：
❌ Array.split 
arr.slice(start , end);   end不包含

❌while(arr1[i] && arr2[j]) 的判断
如果数组里有 `0`，它会被当作 `false` 提前退出循环

❌链表用快慢指针二分，数组用长度/2 二分
